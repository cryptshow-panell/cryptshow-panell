        } else { cell += ch; i++; continue; }

      }

      if (ch === '"') { inQuotes = true; i++; continue; }

      if (ch === delim) { row.push(cell); cell = ""; i++; continue; }

      if (ch === '\r') {

        row.push(cell); rows.push(row); row = []; cell = ""; i++;

        if (i < n && text[i] === '\n') i++;

        continue;

      }

      if (ch === '\n') { row.push(cell); rows.push(row); row = []; cell = ""; i++; continue; }

      cell += ch; i++;

    }

    if (cell.length > 0 || row.length > 0) { row.push(cell); rows.push(row); }

    return rows

      .filter(function (r) {

        if (r.length === 0) return false;

        for (var j = 0; j < r.length; j++) { if (r[j] && r[j].trim() !== "") return true; }

        return false;

      })

      .map(function (r) { return r.map(function (c) { return (c == null ? "" : c).trim(); }); });

  function parsejarDurada(text) {

    var s = (text || "").toString().trim();

    if (!s) return null;

    var m0 = s.match(/^(\d{1,2})\s*[:.]\s*(\d{1,2})\s*[:.]\s*(\d{1,2})$/);

    if (m0) return { min: parseInt(m0[1], 10) * 60 + parseInt(m0[2], 10), sec: parseInt(m0[3], 10) };

    var m1 = s.match(/^(\d+)\s*[:.,]\s*(\d{1,2})$/);

    if (m1) {

      var min1 = parseInt(m1[1], 10), sec1 = parseInt(m1[2], 10);

      if (sec1 > 59) sec1 = Math.round(sec1 / 10);

      return { min: min1, sec: sec1 };

    }

    var m2 = s.match(/^(\d+)\s*['’]\s*(\d{1,2})/);

    if (m2) {

      var min2 = parseInt(m2[1], 10), sec2 = parseInt(m2[2], 10);

      if (sec2 > 59) sec2 = 0;

      return { min: min2, sec: sec2 };

    }

    var m3 = s.match(/(\d+)\s*(?:min|m)\D+?(\d+)\s*(?:s|sec)/i);

    if (m3) return { min: parseInt(m3[1], 10), sec: parseInt(m3[2], 10) };

    var m4 = s.match(/^(\d+)\s*(?:min|m|minuts?)$/i);

    if (m4) return { min: parseInt(m4[1], 10), sec: 0 };

    var m5 = s.match(/^(\d+)$/);

    if (m5) {

      var n = parseInt(m5[1], 10);

      if (n >= 60) return { min: Math.floor(n / 60), sec: n % 60 };

      return { min: n, sec: 0 };

    }

    var m6 = s.match(/^(\d+)\s*h\D*?(\d+)?/i);

    if (m6) {

      var h = parseInt(m6[1], 10), mm = m6[2] ? parseInt(m6[2], 10) : 0;

      return { min: h * 60 + mm, sec: 0 };

    }

    return null;

  }

  function horaAMinuts(h) {

    var p = (h || "").split(":");

    var hh = parseInt(p[0], 10); if (isNaN(hh)) hh = 0;

    var mm = p.length > 1 ? parseInt(p[1], 10) : 0; if (isNaN(mm)) mm = 0;

    return hh * 60 + mm;

  }

  function minutsAHora(mins) {

    mins = ((mins % 1440) + 1440) % 1440;

  }

  function seSolapen(a, b) {

    if (!a.dia || a.dia !== b.dia) return false;

    if (!a.horaInici || !b.horaInici) return false;

    var ai = horaAMinuts(a.horaInici), af = horaFiMinuts(a);

    var bi = horaAMinuts(b.horaInici), bf = horaFiMinuts(b);

    return ai < bf && bi < af;

  }

  function hiHaConflicteCercle(prova, totes, ignorarId) {

    if (!prova.dia || !prova.horaInici) return false;

    for (var i = 0; i < totes.length; i++) {

      var o = totes[i];

      if (o.id === ignorarId) continue;

      if (esDescans(o)) continue;

      if (esDescans(prova)) continue;

      if (o.lloc !== "cercle") continue;

      if (!o.dia || !o.horaInici) continue;

      if (seSolapen(prova, o)) return true;

    }

    return false;

  }

  function teAvisSolapament(s, totes) {

    if (esDescans(s)) return false;

    if (!s.dia || !s.horaInici) return false;

    for (var i = 0; i < totes.length; i++) {

      var o = totes[i];

      if (o.id === s.id) continue;

      if (esDescans(o)) continue;

      if (o.lloc === s.lloc) continue;

      if (!o.dia || !o.horaInici) continue;

      if (seSolapen(s, o)) return true;

    }

    return false;

  }

    var af = ai + (parseInt(a.durMin, 10) || 0);

    var bi = horaAMinuts(b.horaInici || "00:00");

    return ai < bi && af > bi;

  }


  function literariItemsAmbAvis(items) {

    if (!state.esAdmin) {

      return '<section class="csx-panel"><div class="csx-noacces">' +

        '<h2>Accés restringit</h2>' +

        '<p>Aquesta secció només està disponible per a administradors.</p>' +

        '</div></section>';

    }

    var usuarisArr = Object.keys(state.usuaris).map(function (k) {

      var u = state.usuaris[k];

      u._key = k;

      return u;

    }).sort(function (a, b) {

      return (a.email || "").localeCompare(b.email || "");

    });


    var files = usuarisArr.map(function (u) {

      var email = u.email || "";

      var rolsTxt = "";

      if (u.admin === true || esEmailAdmin(email)) {

        rolsTxt = '<span style="color:var(--accent-amber);font-weight:500;">Administrador</span>';

      } else if (u.rols && typeof u.rols === "object") {

        var actius = Object.keys(u.rols).filter(function (k) { return u.rols[k]; });

        if (actius.length === 0) {

          rolsTxt = '<span style="color:var(--text-dim);">Sense rols</span>';

        } else {

          rolsTxt = actius.map(function (k) {

            var r = ROLS_DISPONIBLES.filter(function (x) { return x.id === k; })[0];

            return escapeHtml(r ? r.label : k);

          }).join(", ");

        }

      } else {

        rolsTxt = '<span style="color:var(--text-dim);">Sense rols</span>';

      }

      var esAdminActual = (u.admin === true) || esEmailAdmin(email);

      var dataCreacio = u.creatEl ? new Date(u.creatEl).toLocaleDateString("ca-ES") : "—";

      var accions = '';

      if (!esAdminActual) {

        accions += '<button class="csx-link-btn" data-edit-rol="' + escapeHtml(email) + '" style="margin-right:8px;">canviar rols</button>';

      }

      if (email.toLowerCase() !== (state.usuari.email || "").toLowerCase()) {

        accions += '<button class="csx-link-btn" data-del-user="' + escapeHtml(email) + '">eliminar</button>';

      } else {

        accions += '<span style="font-size:11px;color:var(--text-dim);">(tu)</span>';

      }

      return '<tr>' +

        '<td>' + escapeHtml(email) + '</td>' +

        '<td>' + rolsTxt + '</td>' +

        '<td style="font-size:12px;color:var(--text-dim);">' + escapeHtml(dataCreacio) + '</td>' +

        '<td>' + accions + '</td>' +

        '</tr>';

    }).join("");


    var buida = usuarisArr.length === 0

      ? '<tr><td colspan="4" style="color:var(--text-dim);font-style:italic;padding:14px 10px;">Encara no hi ha usuaris.</td></tr>'

      : "";


    return '<section class="csx-panel">' +

      '<div class="csx-panel-head">' +

      '<div><div class="csx-eyebrow">Gestió d\'usuaris</div><h2>Usuaris</h2></div>' +

      '<button class="csx-btn" id="user-crear">+ Crear usuari</button>' +

      '</div>' +

      '<p style="font-size:13px;color:var(--text-dim);line-height:1.6;margin:-8px 0 20px;">' +

      'Des d\'aquí pots crear comptes per als companys. Els hi has d\'enviar el correu i la contrasenya temporal per un altre canal (WhatsApp, correu, etc.).<br>' +

      'Els usuaris amb <strong style="color:var(--accent-amber);">Administrador</strong> tenen accés total.' +

      '</p>' +

      '<table class="csx-table">' +

      '<thead><tr><th>Correu</th><th>Rols</th><th>Creat el</th><th>Accions</th></tr></thead>' +

      '<tbody>' + files + buida + '</tbody>' +

      '</table>' +

      '</section>';

  }

  function processarCSV(file, seccio) {

    if (!file) return;

    var reader = new FileReader();

    reader.onload = function (e) {

      var rows = parseCSV(String(e.target.result));

      if (rows.length < 1) { window.alert("Fitxer buit."); return; }

      var fc = detectarFilaCapsalera(rows);

      state.csvImport = {

        seccioDesti: seccio, allRows: rows, headerRowIndex: fc,

        headers: rows[fc], rows: rows.slice(fc + 1),

        mapping: autoMapejarColumnes(rows[fc]),

        deteccioAutomatica: true

      };

      render();

    };

    reader.onerror = function () { window.alert("No s'ha pogut llegir."); };

    reader.readAsText(file, "UTF-8");

  }
